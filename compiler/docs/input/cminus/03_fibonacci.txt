/* Entrada: N. Saida: o N-esimo numero de Fibonacci, com F(0) = 0. */

void main(void)
{
    int n;
    int i;
    int previous;
    int current;
    int next;

    n = input();
    i = 0;
    previous = 0;
    current = 1;
    next = 0;

    while (i < n) {
        next = previous + current;
        previous = current;
        current = next;
        i = i + 1;
    }

    output(previous);
    return;
}
