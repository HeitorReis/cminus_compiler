/* Saida: maior valor do vetor fixo. */

int numbers[8];

void main(void)
{
    int i;
    int largest;

    i = 0;
    largest = 0;
    numbers[0] = 12;
    numbers[1] = 7;
    numbers[2] = 19;
    numbers[3] = 3;
    numbers[4] = 25;
    numbers[5] = 4;
    numbers[6] = 18;
    numbers[7] = 9;

    largest = numbers[0];
    i = 1;
    while (i < 8) {
        if (numbers[i] > largest) {
            largest = numbers[i];
        }
        i = i + 1;
    }

    output(largest);
    return;
}
