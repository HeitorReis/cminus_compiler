/* Entrada: dois inteiros positivos. Saida: o maximo divisor comum. */

void main(void)
{
    int a;
    int b;
    int quotient;
    int remainder;

    a = input();
    b = input();
    quotient = 0;
    remainder = 0;

    while (b != 0) {
        quotient = a / b;
        remainder = a - quotient * b;
        a = b;
        b = remainder;
    }

    output(a);
    return;
}
